Обработка столкновений

private bool CheckCollision(Rectangle rect1, Rectangle rect2)
{
    return rect1.IntersectsWith(rect2);
}

private void UpdateGame()
{
    Rectangle playerRect = new Rectangle(playerX, playerY, 20, 20);
    
    foreach (var enemy in enemies)
    {
        Rectangle enemyRect = new Rectangle(enemy.X, enemy.Y, 20, 20);
        if (CheckCollision(playerRect, enemyRect))
        {
            // Столкновение!
            GameOver();
            return;
        }
    }
}

📝 Пример задания — отскакивающий шар

Шар движется с начальной скоростью (4, 3) и отражается от стенок формы. Таймер 16 мс обеспечивает ~60 FPS:

using System;
using System.Drawing;
using System.Windows.Forms;

namespace CSharpGraphicsApp
{
    public class Task16_BouncingBall : Form
    {
        private PointF position = new PointF(100, 100);
        private PointF velocity = new PointF(4, 3);
        private readonly Timer timer = new Timer();

        public Task16_BouncingBall()
        {
            Text = "Задание 18 — Прыгающий шар";
            Size = new Size(600, 400);
            BackColor = Color.White;
            DoubleBuffered = true;
            Paint += Form1_Paint;

            timer.Interval = 16;
            timer.Tick += Timer_Tick;
            timer.Start();
            FormClosed += (s, e) => timer.Stop();
        }

        private void Timer_Tick(object sender, EventArgs e)
        {
            position.X += velocity.X;
            position.Y += velocity.Y;

            if (position.X < 0 || position.X > ClientSize.Width - 50)
                velocity.X = -velocity.X;
            if (position.Y < 0 || position.Y > ClientSize.Height - 50)
                velocity.Y = -velocity.Y;

            position.X = Math.Max(0, Math.Min(position.X, ClientSize.Width - 50));
            position.Y = Math.Max(0, Math.Min(position.Y, ClientSize.Height - 50));

            Invalidate();
        }

        private void Form1_Paint(object sender, PaintEventArgs e)
        {
            e.Graphics.SmoothingMode =
                System.Drawing.Drawing2D.SmoothingMode.AntiAlias;
            e.Graphics.FillEllipse(Brushes.Red,
                position.X, position.Y, 50, 50);
            e.Graphics.DrawString("Отскоки от стенок",
                SystemFonts.DefaultFont, Brushes.Black, 10, 10);
        }
    }
}

Задание по аналогии: Отскакивающий шар

По аналогии с Task16_BouncingBall реализуйте шар, отражающийся от границ окна.

  1. Добавьте поля PointF position и PointF velocity с начальными значениями.
  2. В Timer_Tick обновляйте позицию: position.X += velocity.X, position.Y += velocity.Y.
  3. При касании левой/правой границы инвертируйте velocity.X, при касании верхней/нижней — velocity.Y.
  4. Зафиксируйте позицию через Math.Clamp, чтобы шар не вышел за пределы.
  5. Дополнительно: увеличивайте скорость на 0.1% после каждого отскока.
💡 Совет: Проверяйте границу не > Width, а > Width - ballSize, иначе шар будет «застревать» в стенке.
